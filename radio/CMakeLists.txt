cmake_minimum_required(VERSION 3.16)
project(raspmeshcore_radio_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(raspmeshcore_ch341 STATIC src/Ch341MeshStickTransport.cpp)
target_include_directories(raspmeshcore_ch341 PUBLIC include)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
target_link_libraries(raspmeshcore_ch341 PUBLIC PkgConfig::LIBUSB)
enable_testing()
add_executable(radio_contract_test tests/transport_contract_test.cpp)
target_include_directories(radio_contract_test PRIVATE include)
add_test(NAME radio_contract COMMAND radio_contract_test)
